# \# SPEC: Arquitetura Zero Trust com Dual-Token (Substituição do HMAC)

# 

# \*\*Objetivo:\*\* Alterar o mecanismo de segurança para integrações entre API Rest e MCP (Model Context Protocol).

# \*\*Status:\*\* Aprovado para Implementação.

# \*\*Decisão Arquitetural:\*\* Descontinuação do uso de HMAC (Context Binding) em favor da arquitetura \*\*Dual-Token\*\* (User Context Forwarding + Machine M2M Token) no ecossistema .NET 8.

# 

# \## 1. Contexto e Motivação

# O modelo anterior considerava o uso de HMAC para validação da máquina somado ao token do Azure AD B2C para o contexto do usuário. Devido à complexidade operacional e riscos de gerenciamento de chaves simétricas, a arquitetura evoluiu para o padrão \*\*Dual-Token\*\*.

# Como o Azure B2C não suporta nativamente \*Token Exchange\* (OBO), a API Rest exigirá dois JWTs simultâneos para rotas do MCP:

# 1\. `Authorization: Bearer <Token\_B2C\_Usuario>` (Identidade de Dados).

# 2\. `X-Machine-Token: Bearer <Token\_M2M\_MCP>` (Identidade de Rede/Máquina).

# 

# Esta SPEC define a blindagem de segurança para garantir o isolamento total entre o tráfego do Portal Web (legado) e do MCP.

# 

# \---

# 

# \## 2. Configurações de Ambiente (Whitelist)

# A validação de `Audiences` e `Client IDs` deve ser baseada no padrão \*Options\* e arrays para suportar múltiplos plugins/ambientes.

# 

# \*\*`appsettings.json` (Exemplo de Estrutura):\*\*

# {

# &#x20; "SecuritySettings": {

# &#x20;   "B2CIssuer": "https://seu-b2c.b2clogin.com/tenant/v2.0/",

# &#x20;   "PortalClientId": "client-id-portal-web",

# &#x20;   "McpUserClientId": "client-id-mcp-plugin",

# &#x20;   "McpAllowedMachineClientIds": \[

# &#x20;     "client-id-m2m-chatgpt",

# &#x20;     "client-id-m2m-claude"

# &#x20;   ]

# &#x20; }

# }

#

# \## 3. Pipeline de Autenticação e Esquemas (.NET 8)

# A aplicação possuirá três esquemas distintos para evitar substituição de tokens (\*Token Substitution Attack\*).

# 

# \*\*Instrução de Código (`Program.cs` - Camada AddAuthentication):\*\*

# ```csharp

# var securityConfig = builder.Configuration.GetSection("SecuritySettings");

# var allowedMachineClients = securityConfig.GetSection("McpAllowedMachineClientIds").Get<string\[]>() ?? Array.Empty<string>();

# 

# builder.Services.AddAuthentication(options =>

# {

# &#x20;   // O Portal é o esquema default e de challenge global

# &#x20;   options.DefaultAuthenticateScheme = "PortalAuthScheme";

# &#x20;   options.DefaultChallengeScheme = "PortalAuthScheme";

# })

# .AddJwtBearer("PortalAuthScheme", options =>

# {

# &#x20;   options.TokenValidationParameters = new TokenValidationParameters {

# &#x20;       ValidateAudience = true, ValidAudience = securityConfig\["PortalClientId"],

# &#x20;       ValidateIssuer = true, ValidIssuer = securityConfig\["B2CIssuer"]

# &#x20;   };

# })

# .AddJwtBearer("UserAuthScheme", options =>

# {

# &#x20;   // Esquema que valida o Token do Usuário repassado pelo MCP

# &#x20;   options.TokenValidationParameters = new TokenValidationParameters {

# &#x20;       ValidateAudience = true, ValidAudience = securityConfig\["McpUserClientId"],

# &#x20;       ValidateIssuer = true, ValidIssuer = securityConfig\["B2CIssuer"]

# &#x20;   };

# })

# .AddJwtBearer("MachineAuthScheme", options =>

# {

# &#x20;   // Esquema que valida o Token M2M (Array de Audiences permitidos)

# &#x20;   options.TokenValidationParameters = new TokenValidationParameters {

# &#x20;       ValidateAudience = true, ValidAudiences = allowedMachineClients,

# &#x20;       ValidateIssuer = true, ValidIssuer = securityConfig\["B2CIssuer"]

# &#x20;   };

# &#x20;   

# &#x20;   // Extração customizada do token da máquina no header X-Machine-Token

# &#x20;   options.Events = new JwtBearerEvents {

# &#x20;       OnMessageReceived = context => {

# &#x20;           var machineToken = context.Request.Headers\["X-Machine-Token"].FirstOrDefault();

# &#x20;           if (!string.IsNullOrEmpty(machineToken))

# &#x20;               context.Token = machineToken.Replace("Bearer ", "").Trim();

# &#x20;           return Task.CompletedTask;

# &#x20;       }

# &#x20;   };

# });

# 4\. Políticas de Autorização (Zero Trust Isolation)

# As políticas devem ser configuradas para isolar controladores legados através da FallbackPolicy e forçar a validação dupla para o MCP.

# 

# Instrução de Código (Program.cs - Camada AddAuthorization):

# 

# C#

# // Registro do avaliador customizado

# builder.Services.AddSingleton<IAuthorizationHandler, DualTokenHandler>();

# 

# builder.Services.AddAuthorization(options =>

# {

# &#x20;   // 1. Blindagem do Portal (Bloqueia M2M ou UserAuth de vazar para cá)

# &#x20;   var portalPolicy = new AuthorizationPolicyBuilder()

# &#x20;       .AddAuthenticationSchemes("PortalAuthScheme")

# &#x20;       .RequireAuthenticatedUser()

# &#x20;       .Build();

# 

# &#x20;   options.DefaultPolicy = portalPolicy; // Cobre controllers com \[Authorize] genérico

# &#x20;   options.FallbackPolicy = portalPolicy; // Cobre controllers esquecidos sem anotação

# 

# &#x20;   // 2. Via Expressa Zero Trust para o MCP

# &#x20;   options.AddPolicy("McpDualTokenPolicy", policy =>

# &#x20;   {

# &#x20;       // Força a validação de AMBOS os esquemas na mesma requisição

# &#x20;       policy.AddAuthenticationSchemes("UserAuthScheme", "MachineAuthScheme");

# &#x20;       policy.Requirements.Add(new DualTokenRequirement());

# &#x20;   });

# });

# 4.1 Requirement e Handler (Dual Token)

# C#

# public class DualTokenRequirement : IAuthorizationRequirement { }

# 

# public class DualTokenHandler : AuthorizationHandler<DualTokenRequirement>

# {

# &#x20;   protected override Task HandleRequirementAsync(AuthorizationHandlerContext context, DualTokenRequirement requirement)

# &#x20;   {

# &#x20;       var hasUserIdentity = context.User.Identities.Any(i => i.AuthenticationType == "UserAuthScheme" \&\& i.IsAuthenticated);

# &#x20;       var hasMachineIdentity = context.User.Identities.Any(i => i.AuthenticationType == "MachineAuthScheme" \&\& i.IsAuthenticated);

# 

# &#x20;       // Aprovação estrita: Exige crachá de dados E crachá de rede

# &#x20;       if (hasUserIdentity \&\& hasMachineIdentity)

# &#x20;           context.Succeed(requirement);

# 

# &#x20;       return Task.CompletedTask;

# &#x20;   }

# }

# 5\. Prevenção contra Colisão de Contexto (Context Collision)

# Como o ClaimsPrincipal possuirá múltiplas identidades (M2M e User), o código legado (User.FindFirst("oid")) está sujeito a ler o oid da máquina em vez do investidor.

# 

# Instrução para Agente IA: O plano deve incluir a substituição obrigatória de todas as chamadas de extração de ID do usuário nos Application Services e Controllers para o uso da Extension Method abaixo.

# 

# ClaimsPrincipalExtensions.cs:

# 

# C#

# using System.Security.Claims;

# using System.Linq;

# 

# namespace ZeroTrustApp.Security.Extensions

# {

# &#x20;   public static class ClaimsPrincipalExtensions

# &#x20;   {

# &#x20;       public static string GetInvestorId(this ClaimsPrincipal user)

# &#x20;       {

# &#x20;           if (user == null) return null;

# 

# &#x20;           // Filtro rígido: Só extrai claim de identidades humanas

# &#x20;           var userIdentity = user.Identities.FirstOrDefault(i => 

# &#x20;               i.AuthenticationType == "PortalAuthScheme" || 

# &#x20;               i.AuthenticationType == "UserAuthScheme");

# 

# &#x20;           if (userIdentity == null) return null;

# 

# &#x20;           var oidClaim = userIdentity.FindFirst("\[http://schemas.microsoft.com/identity/claims/objectidentifier](http://schemas.microsoft.com/identity/claims/objectidentifier)") 

# &#x20;                       ?? userIdentity.FindFirst("oid") 

# &#x20;                       ?? userIdentity.FindFirst("sub");

# 

# &#x20;           return oidClaim?.Value;

# &#x20;       }

# &#x20;   }

# }

# 7\. Diretivas de Atualização para o Agente de IA (Plano de Ação)

# Agente, ao ler esta especificação, atualize o plano de execução (execution\_plan) conforme os passos:

# 

# Remoção de Débito: Remover qualquer menção à construção, validação ou infraestrutura de HMAC (incluindo interceptadores ou middlewares criptográficos).

# 

# Atualização do appsettings.json: Incluir o bloco tipado para suporte a múltiplas strings em McpAllowedMachineClientIds.

# 

# Injeção de Dependência: Refatorar a configuração do AddAuthentication para registrar três esquemas (Portal, User, Machine). Incluir o evento para leitura do X-Machine-Token.

# 

# Resiliência Legada: Implementar a FallbackPolicy travada no PortalAuthScheme.

# 

# Implementação do Handler: Adicionar as classes DualTokenRequirement e DualTokenHandler.

# 

# Refatoração de Claims: Substituir globalmente User.FindFirst("oid") por User.GetInvestorId(), adicionando a classe Extension Method estipulada no Item 5.
